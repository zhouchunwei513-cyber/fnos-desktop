// v3.0.51：窗口背景色连刷（赶 DWM 兜底灰）。提到模块作用域，供 win-drag-end 在松手后补刷一轮；

// v2.17.49（用户要求：已最大化的窗口去拖标题栏，应当先还原到最大化前的尺寸和位置，再跟着
// 鼠标走）。这里只登记「待还原目标」，不立刻 unmaximize——否则单击、双击也会把最大化窗口
// 弹下来；等真正开始移动（第一条 win-drag-move）时才还原，且本次不位移，只重设拖动基准。
function __winDragBegin(w, sx, sy) {
  try {
    if (!w || w.isDestroyed()) return null;
    if (__isFsOn(w)) return null; // 全屏态标题栏已隐藏，不允许拖动
    const b = w.getBounds();
    let pending = null;
    if (__maxOn(w) || w.isMaximized()) {
      try { pending = __maxPre(w) || null; } catch (_) { pending = null; }
      if (!pending) { try { pending = w.getNormalBounds(); } catch (_) { pending = null; } }
      if (pending && !(pending.width > 10 && pending.height > 10)) pending = null;
      // v3.0.23：拖动还原的最大化前尺寸同样要归一化——否则「按住顶栏拖窗口」
      // 会把窗口直接撑到 2400x1500（工作区只有 1600x860），看起来就是一直在放大。
      if (pending) pending = __normalizeRestoreBounds(w, pending);
    }
    return {
      at: Date.now(), // v3.0.66：拖动起点时间戳
      win: w, ox: b.x, oy: b.y,
      sx: Number(sx) || 0, sy: Number(sy) || 0,
      lx: b.x, ly: b.y, maxPending: pending,
      // v3.0.23：拖动开始时把尺寸记下来，整个拖动过程只允许改位置、不允许改尺寸
      w0: b.width, h0: b.height
    };
  } catch (_) { return null; }
}
// v3.0.77（用户工单：电视直播窗口拖动不了）——
// 根因：拖动链路是「渲染层收到 mousemove → 发 win-drag-move → 主进程 setBounds」。电视直播窗上
// MPV 是独立的原生窗口，从内容区顶边(y=34)往下整片盖着宿主窗；拖动时窗口移动略滞后于鼠标，
// 鼠标一旦越过 y=34 就落进 MPV 窗口，Chromium 的 renderer 再也收不到 mousemove，拖动当帧就
// 卡死（表现就是「拖不动」）。主窗点播也有同样的覆盖，只是那边窗口跟得上、鼠标不易掉进去。
// 修法：拖动期间由主进程自己按屏幕光标驱动窗口位置（16ms 一帧），完全不依赖 renderer 事件；
// 光标连续静止 600ms、或渲染层报松开，即判定结束。渲染层的 win-drag-move 保留作原路径，
// 两者共用同一 (ox,oy,sx,sy) 基准，不会互相打架。
let __dragPollTimer = null;
function __dragPollStop() {
  try { if (__dragPollTimer) { clearInterval(__dragPollTimer); __dragPollTimer = null; } } catch (_) {}
}
function __dragPollStart(d) {
  __dragPollStop();
  if (!d || !d.win || d.win.isDestroyed()) return;
  let lastX = null, lastY = null, still = 0, ticks = 0, moved = false;
  __dragPollTimer = setInterval(() => {
    try {
      if (__winDragState !== d || !d.win || d.win.isDestroyed()) { __dragPollStop(); return; }
      // 最大化窗口的首帧还原交给渲染层那条 move 处理（它要 unmaximize + 回填拖动基准）
      if (d.maxPending) { still = 0; return; }
      ticks += 1;
      if (ticks < 2) return; // 头一帧放宽：等渲染层的 start 基准落定
      const p = screen.getCursorScreenPoint();
      if (p.x !== lastX || p.y !== lastY) { lastX = p.x; lastY = p.y; still = 0; }
      else { still += 16; }
      // v3.0.79: 只有真的拖起来过（离开起点 >4px），静止阈值才算数。
      // 否则「按下标题栏 -> 停一下 -> 再移动」会被 600ms 静止判成收尾，
      // 后面的 win-drag-move 找不到 __winDragState，表现就是「第一次拖不动，第二次才行」。
      if (!moved && (Math.abs(p.x - d.sx) > 4 || Math.abs(p.y - d.sy) > 4)) moved = true;
      const nx = Math.round(d.ox + p.x - d.sx);
      const ny = Math.round(d.oy + p.y - d.sy);
      if (nx !== d.lx || ny !== d.ly) {
        d.lx = nx; d.ly = ny;
        try {
          if (d.w0 && d.h0) d.win.setBounds({ x: nx, y: ny, width: d.w0, height: d.h0 });
          else d.win.setPosition(nx, ny);
        } catch (_) {}
      }
      if (moved && still >= 600) {
        // 鼠标多半是在 MPV 原生窗上松的手，renderer 收不到 mouseup：这里收尾并让渲染层复位
        __dragPollStop();
        try { if (__winDragState === d) __winDragState = null; } catch (_) {}
        try { __winDragEndAt = Date.now(); } catch (_) {}
        try { if (__ghostBurst) __ghostBurst(); } catch (_) {}
        try { fnosLog('info', 'window', 'drag.poll-end win=' + d.win.id); } catch (_) {}
        try {
          const wc = d.win.webContents;
          if (wc && !wc.isDestroyed()) wc.send('window-drag-cancelled', {});
        } catch (_) {}
      }
    } catch (_) { try { __dragPollStop(); } catch (_) {} }
  }, 16);
  try { if (__dragPollTimer.unref) __dragPollTimer.unref(); } catch (_) {}
}

ipcMain.on('win-drag-start', (e, p) => {
  try {
    const w = hostWindowFromSender(e && e.sender);
    if (!w || w.isDestroyed() || !p) return;
    const st = __winDragBegin(w, p.sx, p.sy);
    if (!st) return;
    __winDragState = st;
    // v3.0.77：以主进程读到的光标为拖动基准（与渲染层 screenX/Y 同为 DIP，但能消除缩放偏差）
    try {
      const cp = screen.getCursorScreenPoint();
      if (cp && Number.isFinite(cp.x) && Number.isFinite(cp.y)) { st.sx = cp.x; st.sy = cp.y; }
    } catch (_) {}
    __dragPollStart(st);
    try {
      fnosLog('info', 'window', 'drag.start win=' + w.id + ' o=' + st.ox + ',' + st.oy
        + ' maxPending=' + (st.maxPending ? [st.maxPending.x, st.maxPending.y, st.maxPending.width, st.maxPending.height].join(',') : '-'));
    } catch (_) {}
  } catch (_) {}
});
ipcMain.on('win-drag-move', (e, p) => {
  try {
    const d = __winDragState;
    if (!d || !p || !d.win || d.win.isDestroyed()) return;
    if (hostWindowFromSender(e && e.sender) !== d.win) return; // 多窗口时只认发起拖动的那个
    // v3.0.66（用户工单：拖动标题栏后点 × 点不中，窗口跟着鼠标一起平移）——
    // 根因链：渲染层的拖动状态在 pointerup 被系统吞掉后残留，之后只要鼠标滑过标题栏，
    // 残留状态就会把每条 move 转发上来，窗口一路粘着鼠标走，× 永远躲手。
    // 这里加一道硬守卫：渲染层报上来的 buttons 为 0（左键没按着）就立刻终止本次拖动。
    // 旧调用方不带 buttons 字段（undefined），保持原行为，不影响旧路径。
    if (typeof p.buttons === 'number' && p.buttons === 0) {
      __winDragState = null;
      try { __dragPollStop(); } catch (_) {}
      try { fnosLog('info', 'window', 'drag.move.reject win=' + d.win.id + ' reason=no-button'); } catch (_) {}
      try { if (__ghostBurst) __ghostBurst(); } catch (_) {}
      return;
    }
    const sx = Number(p.sx) || 0, sy = Number(p.sy) || 0;
    if (d.maxPending) {
      // 这一帧只做还原：unmaximize + 回到最大化前的 x/y/宽/高（与双击还原同一结果），
      // 然后把拖动基准改到还原后的位置，本次不移动，避免窗口瞬间跳到鼠标下面。
      const pb = d.maxPending;
      d.maxPending = null;
      const target = {
        x: Math.round(pb.x), y: Math.round(pb.y),
        width: Math.round(pb.width), height: Math.round(pb.height)
      };
      __maxRestoreAt = Date.now(); // v3.0.23：还原期压制比例锁
      try { __maxInfo.set(d.win, { on: false, pre: null, pseudo: false }); } catch (_) {}
      try { d.win.unmaximize(); } catch (_) {}
      try { d.win.setBounds(target); } catch (_) {}
      d.ox = target.x; d.oy = target.y; d.sx = sx; d.sy = sy; d.lx = target.x; d.ly = target.y;
      d.w0 = target.w
